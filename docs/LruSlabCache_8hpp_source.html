98 const Slab_& find(Id_ id, Cfunction_ create, Pfunction_ populate) {
99 // Avoid a hash lookup if the 'id' is the same as the previous call to find(), e.g., for near-consecutive rows/columns in the same slab.
100 // Note that my_last_slab/id is the same as my_cache_data.back()->first/second, respectively,
101 // but we use separate variables to avoid unnecessarily touching the list data structure.
102 if (my_last_slab && id == my_last_id) {
103 return *my_last_slab;
104 }
105 my_last_id = id;
106
107 auto it = my_cache_exists.find(id);
108 if (it != my_cache_exists.end()) {
109 auto chosen = it->second;
110 my_cache_data.splice(my_cache_data.end(), my_cache_data, chosen); // move to end.
111 my_last_slab = &(chosen->first);
112 return chosen->first;
113 }
114
115 typename std::list<Element>::iterator location;
116 if (my_cache_data.size() < my_max_slabs) {
117 my_cache_data.emplace_back(create(), id);
118 location = std::prev(my_cache_data.end());
119 } else {
120 location = my_cache_data.begin();
121 my_cache_exists.erase(location->second);
122 location->second = id;
123 my_cache_data.splice(my_cache_data.end(), my_cache_data, location); // move to end.
124 }
125 my_cache_exists[id] = location;
126
127 auto& slab = location->first;
128 populate(id, slab);
129 my_last_slab = &slab;
130 return slab;
131 }

151// COMMENT:
152// As tempting as it is to implement an LruVariableSlabCache, this doesn't work out well in practice.
153// This is because the Slab_ objects are re-used, and in the worst case, each object would have to be large enough to fit the largest slab.
154// At this point, we have several options:
155//
156// - Pre-allocate each Slab_ instance to have enough memory to fit the largest slab, in which case the slabs are not variable.
157// - Allow Slab_ instances to grow/shrink their memory allocation according to the size of its assigned slab.
158// This reduces efficiency due to repeated reallocations, and memory usage might end up exceeding the nominal limit anyway due to fragmentation.
159// - Share a single memory pool across slabs, and manually handle defragmentation to free up enough contiguous memory for each new slab.
160// Unlike the oracular case, we don't have the luxury of defragmenting once for multiple slabs.
161// Instead, we might potentially need to defragment on every newly requested slab, which is computationally expensive.
162//
163// See also https://softwareengineering.stackexchange.com/questions/398503/is-lru-still-a-good-algorithm-for-a-cache-with-diferent-size-elements.
164// This lists a few methods for dealing with fragmentation, but none of them are particularly clean.
165// It's likely that just using the existing LruSlabCache with the maximum possible slab size is good enough for most applications.
